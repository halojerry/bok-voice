```python
class TrunkRegisterRequest(BaseModel):
    address: str
    auth_username: str = ""
    auth_password: str = ""
    numbers: list[str] = []

@app.post("/api/sip/sites/{site_id}/trunk")
async def register_sip_trunk(site_id: str, req: TrunkRegisterRequest) -> dict:
    site = _repo().get_site(site_id)
    if not site:
        raise HTTPException(404, "site not found")
    if not req.address or not req.numbers:
        raise HTTPException(400, "address 与 numbers 必填")
    client = _lkapi_client()
    if client is None:
        raise HTTPException(502, "LiveKit 凭据未配置（livekit-sip 不可达）")
    from livekit.api import SIPOutboundTrunkConfig
    try:
        info = await client.sip.create_sip_outbound_trunk(SIPOutboundTrunkConfig(
            name=f"outbound-{site_id[:8]}-{int(_dt.datetime.now(_dt.timezone.utc).timestamp())}",
            address=req.address, numbers=list(req.numbers),
            auth_username=req.auth_username, auth_password=req.auth_password,
            dispatch_ruleless=True,
        ))
    except Exception as exc:
        raise HTTPException(502, f"trunk 注册失败——livekit-sip 未部署或不可达: {exc}")
    finally:
        await client.aclose()
    trunk_id = str(getattr(info, "sip_trunk_id", "") or "")
    updated = _repo().update_site(site_id, trunk_id=trunk_id) or site
    _audit("sip.trunk_registered", subject_type="site", subject_id=site_id,
           account_id=str(site.get("account_id", "acc-001")),
           detail={"trunk_id": trunk_id, "address": req.address})
    return {"trunk_id": trunk_id, "site": updated}
```

```python
def downsample_to_narrowband(pcm16: bytes, src_rate: int = SAMPLE_RATE) -> bytes:
    """模拟电话窄带:16k→8k(相邻两样本均值,顺带抗混叠)→升回 16k(零阶保持)。
    保留 ~0-4kHz 电话频带特征,高频能量被压掉。"""
    import array
    a = array.array("h", pcm16[: len(pcm16) // 2 * 2])
    half = array.array("h", ((a[i] + a[i + 1]) // 2 for i in range(0, len(a) - 1, 2)))
    out = array.array("h")
    for v in half:
        out.append(v); out.append(v)   # 升回原采样率推流(rtc source 16k 固定)
    return out.tobytes()
```
